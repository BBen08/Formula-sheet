\sect{cDer}{Derivatives, Integrals \& Differential Equations}
\expl{$Df(x)$ and $f'(x)$ both mean the derivative. In Dutch, $\arcsin$ and $\arctan$ are often written Bgsin and Bgtan.}

% ---------- Derivatives ----------
\sub{Derivative at a point (definition)}
\[ f'(a) = \lim_{x\to a} \frac{f(x) - f(a)}{x - a} \]

\sub{Basic derivatives: powers}
\[ (c)' = 0 \qquad (x)' = 1 \qquad (x^n)' = n x^{n-1} \qquad \left(\frac{1}{x}\right)' = -\frac{1}{x^2} \]

\sub{Basic derivatives: trigonometric}
\[ (\sin x)' = \cos x \qquad (\cos x)' = -\sin x \qquad (\tan x)' = \frac{1}{\cos^2 x} = \sec^2 x \qquad (\cot x)' = -\frac{1}{\sin^2 x} = -\csc^2 x \]

\sub{Basic derivatives: inverse trigonometric}
\[ (\arcsin x)' = \frac{1}{\sqrt{1 - x^2}} \qquad (\arccos x)' = -\frac{1}{\sqrt{1 - x^2}} \qquad (\arctan x)' = \frac{1}{1 + x^2} \qquad (\operatorname{arccot} x)' = -\frac{1}{1 + x^2} \]

\sub{Basic derivatives: exponential and logarithmic}
\[ (e^x)' = e^x \qquad (a^x)' = a^x \ln a \qquad (\ln x)' = \frac{1}{x} \qquad (\log_a x)' = \frac{1}{x\ln a} \qquad (x^x)' = x^x(\ln x + 1) \]

\sub{Sum, product and quotient rules}
\[ (f \pm g)' = f' \pm g' \qquad (f\cdot g)' = f'g + fg' \qquad \left(\frac{f}{g}\right)' = \frac{f'g - fg'}{g^2} \]

\sub{Chain rule}
\[ \left(f \circ g\right)'(x) = f'\!\left(g(x)\right)\cdot g'(x) \]

\sub{L'Hôpital's rule}
\[ \lim_{x\to a} \frac{f(x)}{g(x)} = \lim_{x\to a} \frac{f'(x)}{g'(x)} \]
\expl{Only for the forms $\frac{0}{0}$ and $\frac{\infty}{\infty}$.}

\sub{Tangent and normal line at a point}
\[ \text{tangent: } y - f(a) = f'(a)(x - a) \qquad \text{normal: } y - f(a) = -\frac{1}{f'(a)}(x - a) \]

\sub{Slope of the normal line}
\[ m_n = -\frac{1}{m_t} \iff m_n\cdot m_t = -1 \]
\expl{$m_t$ = slope of the tangent (Dutch: rico).}

% ---------- Integrals ----------
\sub{Basic integrals: powers}
\[ \int 1\,dx = x + c \qquad \int x^q\,dx = \frac{x^{q+1}}{q+1} + c\ \ (q \ne -1) \qquad \int \frac{1}{x}\,dx = \ln|x| + c \]

\sub{Basic integrals: exponential and logarithmic}
\[ \int e^x\,dx = e^x + c \qquad \int a^x\,dx = \frac{a^x}{\ln a} + c \qquad \int \ln x\,dx = x(\ln x - 1) + c \]

\sub{Basic integrals: trigonometric}
\[ \int \sin x\,dx = -\cos x + c \qquad \int \cos x\,dx = \sin x + c \qquad \int \sec^2 x\,dx = \tan x + c \qquad \int \csc^2 x\,dx = -\cot x + c \]
\[ \int \tan x\,dx = \ln|\sec x| + c \qquad \int \cot x\,dx = \ln|\sin x| + c \]
\[ \int \sec x\,dx = \ln|\sec x + \tan x| + c \qquad \int \csc x\,dx = -\ln|\csc x + \cot x| + c \]

\sub{Basic integrals: giving arcsin, arctan and ln}
\[ \int \frac{dx}{\sqrt{1 - x^2}} = \arcsin x + c \qquad \int \frac{dx}{1 + x^2} = \arctan x + c \]
\[ \int \frac{dx}{\sqrt{a^2 - x^2}} = \arcsin\frac{x}{a} + c \qquad \int \frac{dx}{x^2 + a^2} = \frac{1}{a}\arctan\frac{x}{a} + c \qquad \int \frac{dx}{\sqrt{x^2 + k}} = \ln\left|x + \sqrt{x^2 + k}\right| + c \]

\sub{Linearity of the integral}
\[ \int \left(r f(x) \pm s\,g(x)\right)dx = r\int f(x)\,dx \pm s\int g(x)\,dx \]

\sub{Integration by substitution}
\[ \int f(ax + b)\,dx = \frac{1}{a}\int f(ax + b)\,d(ax + b) \]

\sub{Integration by parts}
\[ \int u\,dv = uv - \int v\,du \]

\sub{Integration by partial fractions}
\[ \int \frac{A}{ax + b}\,dx = \frac{A}{a}\ln|ax + b| + c \qquad \int \frac{B}{(ax + b)^n}\,dx = \frac{B}{a}\cdot\frac{(ax + b)^{-n+1}}{-n+1} + c \]
\expl{Quadratic denominator with $D < 0$: $\int \frac{C}{ax^2 + bx + c}\,dx$ gives an $\arctan$; $\int \frac{dx + e}{ax^2 + bx + c}\,dx$ gives $\ln|\dots| + \arctan(\dots)$.}

\sub{Definite integral (Riemann sum)}
\[ \int_a^b f(x)\,dx = \lim_{n\to +\infty} \sum_{i=1}^{n} f(x_i)\,\Delta x \]

\sub{Properties of the definite integral}
\[ \int_a^c f(x)\,dx + \int_c^b f(x)\,dx = \int_a^b f(x)\,dx \]
\[ \int_a^b \left(r f(x) + s\,g(x)\right)dx = r\int_a^b f(x)\,dx + s\int_a^b g(x)\,dx \]

\sub{Integral function}
\[ I_a(x) = \int_a^x f(t)\,dt \]

\sub{Mean value theorem for integrals}
\[ \int_a^b f(x)\,dx = (b - a)\,f(c) \qquad f(c) = \frac{1}{b - a}\int_a^b f(x)\,dx \]
\expl{$f(c)$ is the average value of $f$ on $[a, b]$.}

\sub{Fundamental theorem of calculus}
\[ \frac{d}{dx}\int_a^x f(t)\,dt = f(x) \qquad \int_a^b f(x)\,dx = \Big[F(x)\Big]_a^b = F(b) - F(a) \]
\[ \int_a^b f'(x)\,dx = f(b) - f(a) \]

\sub{Volume of a solid of revolution}
\[ \text{about the } x\text{-axis: } V = \pi\int_a^b \left[f(x)\right]^2 dx \qquad \text{about the } y\text{-axis: } V = \pi\int_c^d \left[x(y)\right]^2 dy \]
\[ \text{between two curves: } V = \pi\int_a^b \left(f(x)^2 - g(x)^2\right)dx \]

\sub{Arc length of a curve}
\[ L = \int_a^b \sqrt{1 + \left[f'(x)\right]^2}\,dx \]

\sub{Surface area of a solid of revolution}
\[ \text{about the } x\text{-axis: } S = 2\pi\int_a^b f(x)\sqrt{1 + f'(x)^2}\,dx \qquad \text{about the } y\text{-axis: } S = 2\pi\int_a^b x\sqrt{1 + f'(x)^2}\,dx \]

\sub{Double integral (Fubini's theorem)}
\[ V = \iint_D f(x, y)\,dA = \int_c^d \left(\int_a^b f(x, y)\,dx\right)dy = \int_a^b \left(\int_c^d f(x, y)\,dy\right)dx \]

% ---------- Differential equations ----------
\sub{First-order separable DE}
\[ \frac{dy}{dt} = f(t)\,g(y) \;\Rightarrow\; \int \frac{dy}{g(y)} = \int f(t)\,dt \]

\sub{First-order linear DE and integrating factor}
\[ y' + p(t)\,y = q(t) \qquad \mu(t) = e^{\int p(t)\,dt} \qquad y(t) = \frac{1}{\mu(t)}\left(\int \mu(t)\,q(t)\,dt + C\right) \]

\sub{Second-order linear DE}
\[ a(t)\,y'' + b(t)\,y' + c(t)\,y = r(t) \]

\sub{Homogeneous second-order DE: characteristic equation}
\[ ay'' + by' + cy = 0 \;\Rightarrow\; a\lambda^2 + b\lambda + c = 0 \qquad D = b^2 - 4ac \]

\sub{Homogeneous second-order DE: general solution}
\[ D > 0: \quad y(t) = c_1 e^{\lambda_1 t} + c_2 e^{\lambda_2 t} \]
\[ D = 0: \quad y(t) = c_1 e^{\lambda t} + c_2\,t\,e^{\lambda t} \]
\[ D < 0: \quad y(t) = e^{\alpha t}\left(c_1\cos\beta t + c_2\sin\beta t\right) \qquad \lambda = \alpha \pm \beta i \]
