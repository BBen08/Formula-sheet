\sect{cAlg}{Algebra \& Equations}
\expl{Rules for roots, powers and logarithms, and solving equations. Complex numbers are in Calculus.}

% ---------- Basic algebra ----------
\sub{Absolute value}
\[ |a| = \begin{cases} a & \text{if } a \ge 0 \\ -a & \text{if } a < 0 \end{cases} \]

\sub{Properties of roots}
\[ \sqrt[n]{ab} = \sqrt[n]{a}\cdot\sqrt[n]{b} \qquad \sqrt[n]{\frac{a}{b}} = \frac{\sqrt[n]{a}}{\sqrt[n]{b}} \]
\[ \sqrt[n]{a^m} = \left(\sqrt[n]{a}\right)^m \qquad \sqrt[n]{a^n} = a \qquad \sqrt[m]{a} = a^{\frac{1}{m}} \]

\sub{Properties of powers}
\[ a^{\frac{m}{n}} = \left(a^m\right)^{\frac{1}{n}} = \sqrt[n]{a^m} \]

\sub{Special products}
\expl{In Dutch: merkwaardige producten.}
\[ (a \pm b)^2 = a^2 \pm 2ab + b^2 \qquad a^2 - b^2 = (a+b)(a-b) \]
\[ a^3 \pm b^3 = (a \pm b)\left(a^2 \mp ab + b^2\right) \]
\[ (a \pm b)^3 = a^3 \pm 3a^2b + 3ab^2 \pm b^3 \]

% ---------- Logarithms & exponentials ----------
\sub{Definition of the logarithm}
\[ \log_a x = y \iff a^y = x \qquad \log_a a^y = y \qquad a^{\log_a x} = x \]

\sub{Properties of logarithms}
\[ \log_a (x_1 x_2) = \log_a x_1 + \log_a x_2 \qquad \log_a \frac{x_1}{x_2} = \log_a x_1 - \log_a x_2 \]
\[ \log_a \frac{1}{x} = -\log_a x \qquad \log_a x^n = n \log_a x \]

\sub{Change of base}
\[ \log_b x = \frac{\log_a x}{\log_a b} \qquad \log_b a = \frac{1}{\log_a b} \]
\[ \log_{b^n} a^m = \frac{m}{n}\log_b a \qquad \log_a b^n = \log_{a^{1/n}} b \qquad \log_{a^m} b = \log_a b^{\frac{1}{m}} \]

% ---------- Equations ----------
\sub{Quadratic equation (abc-formula)}
\[ ax^2 + bx + c = 0 \;\Rightarrow\; x_{1,2} = \frac{-b \pm \sqrt{D}}{2a} \qquad D = b^2 - 4ac \]

\sub{Sum and product of the roots}
\[ S = x_1 + x_2 = -\frac{b}{a} \qquad P = x_1 \cdot x_2 = \frac{c}{a} \]

\sub{Euclidean division of polynomials}
\[ A(x) = Q(x)\cdot D(x) + R(x) \qquad \frac{A(x)}{D(x)} = Q(x) + \frac{R(x)}{D(x)} \]
\expl{$A$ = dividend, $D$ = divisor, $Q$ = quotient, $R$ = remainder (degree of $R$ < degree of $D$).}
