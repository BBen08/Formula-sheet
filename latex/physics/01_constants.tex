\sect{cCon}{Physical Constants}
\expl{The unit is given in square brackets after each value.}

\sub{Speed of light}
\[ c = 2.998\cdot 10^{8}\ \mathrm{m/s} = 299\,792\ \mathrm{km/s} \]

\sub{Elementary charge (proton and electron)}
\[ e = 1.602\cdot 10^{-19}\ \mathrm{C} \qquad q_{p^+} = +e \qquad q_{e^-} = -e \]

\sub{Gravitational field strength (Earth)}
\[ g = 9.81\ \mathrm{N/kg} = 9.81\ \mathrm{m/s^2} \]

\sub{Gravitational constant}
\[ G = 6.674\cdot 10^{-11}\ \mathrm{N\,m^2/kg^2} \]

\sub{Ideal gas constant}
\[ R = 8.314\ \mathrm{J/(mol\cdot K)} \]

\sub{Avogadro's number}
\[ N_A = 6.022\cdot 10^{23}\ \mathrm{mol^{-1}} \]

\sub{Permittivity of vacuum}
\[ \varepsilon_0 = 8.85\cdot 10^{-12}\ \mathrm{C^2/(N\,m^2)} \]

\sub{Permeability of vacuum}
\[ \mu_0 = 1.257\cdot 10^{-6}\ \mathrm{T\,m/A} \]

\sub{Electronvolt}
\[ 1\ \mathrm{eV} = 1.602\cdot 10^{-19}\ \mathrm{J} \]

\sub{Atomic mass unit}
\[ 1\ \mathrm{u} = 1.661\cdot 10^{-27}\ \mathrm{kg} \qquad 1\ \mathrm{u}\cdot c^2 = 931.49\ \mathrm{MeV} \]

\sub{Speed of sound}
\[ v_{\text{air}} = 343\ \mathrm{m/s} \qquad v_{\text{water}} \approx 1500\ \mathrm{m/s} \]
